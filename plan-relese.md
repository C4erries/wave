  set -e

  # mbctl create-topic raw.gen.chA, spectrum.gen.chA, raw.osc.chA, spectrum.osc.chA

  echo "Starting Python components in background..."
  source components/integration/.venv/bin/activate
  wave-gen --waveform=sine --freq=1000 &
  echo $! > /tmp/wave-gen.pid
  wave-fft --input-topic=raw.gen.chA --output-topic=spectrum.gen.chA &
  echo $! > /tmp/wave-fft-gen.pid
  wave-fft --input-topic=raw.osc.chA --output-topic=spectrum.osc.chA \
           --group-id=fft-osc-chA &
  echo $! > /tmp/wave-fft-osc.pid
  
  echo "Demo stack ready."
  echo "Open http://localhost:5173/lab"
  echo "Run scripts/stop.sh to stop everything."

  #!/bin/bash
  for pidfile in /tmp/wave-*.pid; do
    [ -f "$pidfile" ] && kill $(cat "$pidfile") 2>/dev/null
    rm -f "$pidfile"
  done
  docker compose -f deploy/docker-compose.demo.yml down

## Несколько коротких замечаний к плану

**Про GUI-адаптер.** tkinter оправдан как «достаточный, не отвлекает от основной работы». Если в процессе попробуешь — и тебе захочется лучше — есть `customtkinter` (одна зависимость, выглядит современно). Но это уже косметика, не сегодня.

**Про оркестратор завтра (пункт 2).** Прежде чем стартовать — реши, нужен ли HTTP API оркестратора, или хватит CLI-команд. HTTP API нужен **только** если ты сегодня вечером решаешь делать пункт 3 (конструктор графа в UI). Если решаешь не делать — оркестратор может быть чисто CLI, что вдвое проще.

**Про пункт 3 (визуальный конструктор).** Если хочешь его сделать — спроси препода или гугли что-то простое типа `react-flow` (он есть, активно поддерживается, drag-and-drop узлы и стрелки). Это **сильнейший** элемент в UI, но 4-6 часов работы агента минимум. Решай завтра.
